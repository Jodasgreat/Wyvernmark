	fire_only_once = yes
	
	option = {
		name = wyvernmark.200.a
		if = {
			limit = { NOT = { is_elector = yes } } 
			if = {
				limit = { any_subject_country = { is_elector = yes } } 
				random_subject_country = {
					limit = { is_elector = yes } 
					elector = no
				}
				elector = yes
			}
			else_if = {
				limit = { A11 = { is_elector = yes } } 
				A11 = { elector = no } 
				elector = yes
			}
			else = {
				random_elector = {
					limit = {
						NOT = {
							tag = A46
							tag = A31
							tag = A45
							tag = A85
							tag = A73
							tag = A12
						}
						ai = yes
					}
					elector = no
				}
				elector = yes
			}
		}
		set_government_rank_scripted_effect = { rank = 2 }
		add_country_modifier = { 
			name = A3A_arch_margravine
			duration = -1
		}
		set_country_flag = A3A_can_integrate
	}
}

country_event = {
	id = wyvernmark.1
	title = wyvernmark.1.t
	desc = wyvernmark.1.d
	picture = muslimgfx_NEW_HEIR_eventPicture
	
	trigger = {
		AND = {
			primary_culture = bronzewing_harpy
			has_country_modifier = monstrous_nation
		}
	}
	
	hidden = yes
	
	option = {
		name = wyvernmark.1.a
		ai_chance = { factor = 100 }
		remove_country_modifier = monstrous_nation
		hidden_effect = {
			every_country = {
				if = { 
					limit = { 
						has_opinion_modifier = { 
							who = ROOT 
							modifier = root_monstrous 
						} 
					}
					remove_opinion = { 
						who = ROOT 
						modifier = root_monstrous 
					}
					reverse_remove_opinion = { 
						who = ROOT 
						modifier = root_monstrous 
						}
				}
				else_if = { 
					limit = { 
						has_opinion_modifier = { 
							who = ROOT 
							modifier = semi_monstrous 
						} 
					}
					remove_opinion = { 
						who = ROOT 
						modifier = semi_monstrous 
					}
					reverse_remove_opinion = { 
						who = ROOT 
						modifier = semi_monstrous 
					}
				}
				else_if = {
					limit = { 
						has_opinion_modifier = { 
							who = ROOT 
							modifier = both_monstrous 
						} 
					}
					remove_opinion = { who = ROOT modifier = both_monstrous }
					reverse_remove_opinion = { who = ROOT modifier = both_monstrous }
				}
			}
		remove_country_modifier = setup_monstrous_opinion
		}
	}
}
